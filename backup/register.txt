# ==========================================================
# BLUEPRINT INITIALIZER
# ==========================================================
def init_register_blueprint(mail):
    register_blueprint = Blueprint("register", __name__)

    # ======================================================
    # 1. REGISTER
    # ======================================================
    @register_blueprint.route("/register", methods=["POST"])
    def register_user():
        try:
            data = request.get_json(silent=True)
            if not data:
                return jsonify({"status": "error", "message": "Request tidak valid"}), 400

            email = str(data.get("email", "")).strip().lower()
            password = str(data.get("password", ""))
            username = data.get("username")
            fullname = data.get("fullname")
            province = data.get("provinsi")
            role = data.get("role", "user")
            image = data.get("image", "default_profile.png")

            if not all([email, password, username, fullname]):
                return jsonify({"status": "error", "message": "Semua field wajib diisi"}), 400

            # Supabase Auth
            try:
                auth = supabase.auth.sign_up({"email": email, "password": password})
                if not auth.user:
                    return jsonify({"status": "error", "message": "Gagal create auth user"}), 400
            except AuthApiError as e:
                if "User already registered" in str(e):
                    return jsonify({"status": "error", "message": "Email sudah terdaftar"}), 409
                raise e 

            # Data Prep
            otp_code = generate_otp()
            otp_expired_at = (datetime.datetime.now(timezone.utc) + datetime.timedelta(minutes=10)).isoformat()
            hashed_password = bcrypt.generate_password_hash(password).decode("utf-8")
            
            payload = {
                "user_id": auth.user.id,
                "email": email,
                "exp": datetime.datetime.now(timezone.utc) + datetime.timedelta(days=7)
            }
            token = jwt.encode(payload, SECRET_KEY, algorithm="HS256")

            user_data = {
                "id": auth.user.id,
                "username": username,
                "fullname": fullname,
                "email": email,
                "province": province,
                "password": hashed_password,
                "otp": int(otp_code),
                "otp_expired_at": otp_expired_at,
                "token": token,
                "role": role,
                "points": 0,
                "image": image,
                "is_verified": False
            }

            supabase.table("users").insert(user_data).execute()
            ActivityLogger.log(f"Registrasi berhasil ({email})", auth.user.id)

            # Send Email
            try:
                msg = Message(subject="Verifikasi Akun Scoutify", recipients=[email])
                msg.html = get_email_html(fullname, otp_code)
                mail.send(msg)
            except Exception as e:
                print(f"[MAIL ERROR] {e}")

            return jsonify({"status": "success", "message": "Registrasi berhasil, cek email", "token": token}), 201

        except Exception as e:
            traceback.print_exc()
            return jsonify({"status": "error", "message": "Server error", "error": str(e)}), 500

    # ======================================================
    # 2. VERIFY OTP
    # ======================================================
    @register_blueprint.route("/verify-otp", methods=["POST"])
    def verify_otp():
        try:
            data = request.get_json(silent=True)
            email = str(data.get("email", "")).strip().lower()
            otp_input = str(data.get("otp", ""))

            res = supabase.table("users").select("*").eq("email", email).execute()
            if not res.data:
                return jsonify({"status": "error", "message": "User tidak ditemukan"}), 404

            user = res.data[0]

            if str(user["otp"]) != otp_input:
                ActivityLogger.log("Verifikasi OTP gagal (kode salah)", user["id"])
                return jsonify({"status": "error", "message": "OTP salah"}), 400

            expired = datetime.datetime.fromisoformat(user["otp_expired_at"].replace("Z", "+00:00"))
            if datetime.datetime.now(timezone.utc) > expired:
                return jsonify({"status": "error", "message": "OTP expired"}), 400

            supabase.table("users").update({
                "is_verified": True,
                "otp": None,
                "otp_expired_at": None
            }).eq("id", user["id"]).execute()

            # PERBAIKAN: Kirim juga data user yang aman ke Frontend
            safe_user_data = {
                "id": str(user["id"]),
                "username": user.get("username", ""),
                "fullname": user.get("fullname", ""),
                "email": user.get("email", ""),
                "role": user.get("role", "user"),
                "province": user.get("province", ""),
                "points": user.get("points", 0),
                "image": user.get("image", "default_profile.png")
            }

            return jsonify({
                "status": "success", 
                "message": "Akun berhasil diverifikasi",
                "token": user["token"],
                "user": safe_user_data # Data ini yang dibutuhkan SessionManager di Flutter
            }), 200

        except Exception as e:
            traceback.print_exc()
            return jsonify({"status": "error", "message": "Server error"}), 500

    # ======================================================
    # 3. RESEND OTP
    # ======================================================
    @register_blueprint.route("/resend-otp", methods=["POST"])
    def resend_otp():
        try:
            data = request.get_json(silent=True)
            email = str(data.get("email", "")).strip().lower()
            
            res = supabase.table("users").select("id, fullname").eq("email", email).execute()
            if not res.data:
                return jsonify({"status": "error", "message": "User tidak ditemukan"}), 404
            
            user = res.data[0]
            new_otp = generate_otp()
            new_expired = (datetime.datetime.now(timezone.utc) + datetime.timedelta(minutes=10)).isoformat()

            supabase.table("users").update({
                "otp": int(new_otp),
                "otp_expired_at": new_expired
            }).eq("id", user["id"]).execute()

            msg = Message(subject="OTP Baru Scoutify", recipients=[email])
            msg.html = get_email_html(user["fullname"], new_otp)
            mail.send(msg)

            return jsonify({"status": "success", "message": "OTP baru terkirim"}), 200
        except Exception as e:
            traceback.print_exc()
            return jsonify({"status": "error", "message": "Gagal resend OTP"}), 500

    return register_blueprint
