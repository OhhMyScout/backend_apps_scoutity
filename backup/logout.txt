import os
from flask import Blueprint, request, jsonify
from config.database import get_supabase_client
from dotenv import load_dotenv

load_dotenv()

# ==========================================================
# SETUP & CONFIGURATION
# ==========================================================
supabase = get_supabase_client()


# ==========================================================
# ACTIVITY LOG HELPER
# ==========================================================
def save_activity_log(user_id=None, activity="", ip_address=None, user_agent=None):
    try:
        supabase.table("activity_logs").insert({
            "user_id": user_id,
            "activity": activity,
            "ip_address": ip_address,
            "user_agent": user_agent
        }).execute()
    except Exception as e:
        print(f"[ACTIVITY LOG ERROR] {str(e)}")


# ==========================================================
# BLUEPRINT INITIALIZER
# ==========================================================
def init_logout_blueprint():
    logout_bp = Blueprint('logout', __name__)

    # ======================================================
    # LOGOUT ENDPOINT
    # ======================================================
    @logout_bp.route('/logout', methods=['POST'])
    def logout_user():
        try:
            # 1. Ambil token JWT dari Header Authorization
            auth_header = request.headers.get('Authorization')
            
            if not auth_header or not auth_header.startswith('Bearer '):
                return jsonify({
                    "status": "error",
                    "message": "Token tidak ditemukan atau format tidak valid"
                }), 401
            
            # Ekstrak string token setelah teks "Bearer "
            token = auth_header.split(" ")[1].strip()
            
            # Get request metadata untuk logging
            ip_addr = request.remote_addr
            user_agnt = request.headers.get('User-Agent', 'Unknown Device')

            # ==================================================
            # CARI USER BERDASARKAN TOKEN
            # ==================================================
            user_check = supabase.table('users') \
                .select('id', 'username') \
                .eq('token', token) \
                .execute()
            
            # Validasi: Jika data kosong, berarti session tidak valid atau sudah logout sebelumnya
            if not user_check.data or len(user_check.data) == 0:
                save_activity_log(
                    activity="Percobaan logout gagal: Token tidak valid/sudah expired",
                    ip_address=ip_addr,
                    user_agent=user_agnt
                )
                return jsonify({
                    "status": "error",
                    "message": "Sesi login tidak valid atau Anda sudah logout"
                }), 404
            
            user_id = user_check.data[0]['id']
            username = user_check.data[0]['username']

            # ==================================================
            # HAPUS TOKEN DI TABEL USERS (SET NULL)
            # ==================================================
            supabase.table('users') \
                .update({"token": None}) \
                .eq('id', user_id) \
                .execute()

            # ==================================================
            # PENCATATAN LOG AKTIVITAS (SUKSES)
            # ==================================================
            save_activity_log(
                user_id=user_id,
                activity=f"Logout sukses ({username})",
                ip_address=ip_addr,
                user_agent=user_agnt
            )

            # ==================================================
            # RESPONSE SUKSES
            # ==================================================
            return jsonify({
                "status": "success",
                "message": "Logout berhasil dilakukan"
            }), 200

        except Exception as e:
            print(f"[LOGOUT ERROR] {str(e)}")
            
            # Ambil metadata request secara aman jika konteks request masih tersedia
            try:
                ip_addr = request.remote_addr
                user_agnt = request.headers.get("User-Agent")
            except Exception:
                ip_addr = None
                user_agnt = None

            save_activity_log(
                activity=f"Server error logout: {str(e)}",
                ip_address=ip_addr,
                user_agent=user_agnt
            )

            return jsonify({
                "status": "error",
                "message": "Terjadi kesalahan server saat proses logout"
            }), 500

    return logout_bp
